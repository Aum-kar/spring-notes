package abyss.WebTwilight;

import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;

import java.util.ArrayList;

@RestController
@RequestMapping("/playground")
class PlaygroundController {

    ArrayList<User> userDatabase = new ArrayList<>();

    public PlaygroundController() {
        userDatabase.add(new User(1, "Omkar"));
        userDatabase.add(new User(2, "Saurabh"));
        userDatabase.add(new User(3, "Shivraj"));
        userDatabase.add(new User(4, "Shubham"));
    }

    @GetMapping(path = "/get-all")
    public ResponseEntity<String> getAllUsers() {
//        return ResponseEntity.status(200).body(userDatabase);
        return ResponseEntity.status(500).body("Unimplemented method");
    }

    @GetMapping("/get-user/{id}")
    public ResponseEntity<User> getUser(@PathVariable int id) {
        for (User user : userDatabase) {
            if(user.getUsername() == id) {
                return ResponseEntity.status(200).body(user);
            }
        }
        return ResponseEntity.status(404).body(new User(-1, null));
    }

    @PostMapping("/add-user")
    public ResponseEntity<String> addUser(User user) {
        userDatabase.add(user);
        return ResponseEntity.status(202).body("User created");
    }

    @DeleteMapping("delete")
    public ResponseEntity<String> deleteUser(User user) {
        userDatabase.remove(user);
        return ResponseEntity.status(200).body("User removed");
    }

}
